        function renderSlots(slots) {
            const grid = document.getElementById('slot-grid');
            const durasi = parseInt(document.getElementById('durasi-input').value) || 1;
            grid.innerHTML = '';

                    div.className =
                        'slot-taken py-sm text-center rounded-DEFAULT relative overflow-hidden opacity-50 line-through cursor-not-allowed';
                    div.innerHTML = `${slot.jam}<div class="absolute inset-0 bg-error/10"></div>`;
                } else {
                    const isSelected = selJam === slot.jam;
                    const isDuration = isDurationSlot(slot.jam, selJam, durasi, slots);
                    if (isSelected) {
                        div.className =
                            'slot-selected py-sm text-center rounded-DEFAULT cursor-pointer relative overflow-hidden group bg-primary text-on-primary font-bold';
                        div.innerHTML =
                            `<div class="absolute inset-0 w-full h-full bg-gradient-to-b from-white/20 to-transparent opacity-0 group-hover:opacity-100 transition-opacity"></div>${slot.jam}`;
                    } else if (isDuration) {
                        div.className =
                            'slot-duration py-sm text-center rounded-DEFAULT border border-primary/50 cursor-pointer bg-primary-container text-on-primary-container';
                        div.textContent = slot.jam;
                    } else {
                        div.className =
                            'py-sm text-center rounded-DEFAULT border border-outline-variant/30 text-on-surface-variant cursor-pointer hover:border-primary/50 hover:text-primary transition-colors';
                        div.textContent = slot.jam;
                    }
                    div.addEventListener('click', () => {
                        selJam = slot.jam;
                        document.getElementById('jam_mulai').value = slot.jam;
                        renderSlots(slots);
                        updateSummary();
                    });
                }
                grid.appendChild(div);
            });
        }

        function isDurationSlot(jam, startJam, durasi, slots) {
            if (!startJam || durasi <= 1) return false;
            const allJams = slots.filter(s => s.available).map(s => s.jam);
            const startIndex = allJams.indexOf(startJam);
            const currentIdx = allJams.indexOf(jam);
            return currentIdx > startIndex && currentIdx < startIndex + durasi;
        }

            const info = document.getElementById('slot-info');
            const loading = document.getElementById('slot-loading');
            const libur = document.getElementById('slot-libur');

            if (info) info.classList.toggle('hidden', state !== 'info');
            if (loading) loading.classList.toggle('hidden', state !== 'loading');
            if (libur) libur.classList.toggle('hidden', state !== 'libur');
            if (grid) grid.classList.toggle('hidden', state !== 'grid');

            if (selectedFacilities.length > 0) {

                    subtotal += itemTotal;

                summaryContainer.innerHTML = facilitiesHtml;

                if (IS_GURU) {
                    total = 0;
                    document.getElementById('sum-subtotal').innerHTML = fmt(subtotal);
                    document.getElementById('sum-total').innerHTML = fmt(0);

                    total = subtotal - totalDiskon;
                    document.getElementById('sum-subtotal').innerHTML = fmt(subtotal);
                    document.getElementById('sum-total').innerHTML = fmt(total);
                    const diskonRow = document.getElementById('diskon-row');

                        if (diskon > 0) {
                            diskonRow.style.removeProperty('display');

                        } else {
                            diskonRow.style.setProperty('display', 'none', 'important');
                        }
                    }
                }
                totalHargaFinal = total;
            } else {
                summaryContainer.innerHTML =
                    '<div class="text-center text-on-surface-variant text-sm py-2">Belum ada fasilitas</div>';
                document.getElementById('sum-subtotal').innerHTML = 'Rp 0';
                document.getElementById('sum-total').innerHTML = 'Rp 0';
                totalHargaFinal = 0;
            }

            // Update tanggal, waktu, durasi
            if (tanggal && document.getElementById('sum-tanggal')) {
                const d = new Date(tanggal);
                const days = ['Min', 'Sen', 'Sel', 'Rab', 'Kam', 'Jum', 'Sab'];
                const months = ['Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun', 'Jul', 'Agu', 'Sep', 'Okt', 'Nov', 'Des'];
                document.getElementById('sum-tanggal').textContent =
                    `${days[d.getDay()]}, ${d.getDate()} ${months[d.getMonth()]} ${d.getFullYear()}`;
            }
            if (selJam && document.getElementById('sum-waktu')) {
                const [h, m] = selJam.split(':').map(Number);
                const endH = h + durasi;
                document.getElementById('sum-waktu').textContent =
                    `${selJam} – ${String(endH).padStart(2, '0')}:${String(m).padStart(2, '0')}`;
            }
            document.getElementById('sum-durasi').textContent = `${durasi} Jam`;
        }

        // Durasi buttons
        const durMinus = document.getElementById('dur-minus');
        const durPlus = document.getElementById('dur-plus');
        const durasiInput = document.getElementById('durasi-input');

        function onDurasiChange() {
            if (selJam) loadSlots();
            updateSummary();
        }

        if (durMinus) durMinus.addEventListener('click', () => {
            if (parseInt(durasiInput.value) > 1) {
                durasiInput.value = parseInt(durasiInput.value) - 1;
                onDurasiChange();
            }
        });
        if (durPlus) durPlus.addEventListener('click', () => {
            if (parseInt(durasiInput.value) < 8) {
                durasiInput.value = parseInt(durasiInput.value) + 1;
                onDurasiChange();
            }
        });
        if (durasiInput) durasiInput.addEventListener('change', onDurasiChange);

        // Date change
        const bookingDate = document.getElementById('booking-date');
        if (bookingDate) {
            bookingDate.addEventListener('change', () => {
                selJam = null;
                document.getElementById('jam_mulai').value = '';
                loadSlots();
                updateSummary();
            });
        }

        // Hide payment methods for GURU
        if (IS_GURU) {
            const metodeSection = document.querySelector('#metode-bayar-section');
            if (metodeSection) metodeSection.style.display = 'none';
            const cashRadio = document.querySelector('input[name="metode_bayar"][value="cash"]');
            if (cashRadio) cashRadio.checked = true;
        }

        // Submit button
        const bookingForm = document.getElementById('booking-form');
        const submitBtn = document.getElementById('submit-btn');

        if (submitBtn) {
            submitBtn.addEventListener('click', function(e) {
                e.preventDefault();
                if (selectedFacilities.length === 0) {
                    showToast('Pilih minimal satu fasilitas.', 'error');
                    return;
                }
                if (!selJam) {
                    showToast('Pilih jam mulai terlebih dahulu.', 'error');
                    return;
                }
                const metode = document.querySelector('input[name="metode_bayar"]:checked')?.value;
                if (!metode) {
                    showToast('Pilih metode pembayaran', 'error');
                    return;
                }
                if (metode === 'cash') {
                    submitBtn.disabled = true;
                    submitBtn.innerHTML =
                        '<span class="material-symbols-outlined animate-spin text-[20px]">progress_activity</span> Memproses...';
                    closePaymentModal();
                    const formData = new FormData(bookingForm);
                    fetch(bookingForm.action, {
                        method: 'POST',
                        body: formData,
                        headers: {
                            'X-CSRF-TOKEN': document.querySelector('input[name="_token"]').value
                        }
                    }).then(response => response.json()).then(data => {
                        if (data.success && data.redirect) window.location.href = data.redirect;
                        else if (data.error) {
                            showToast(data.error, 'error');
                            submitBtn.disabled = false;
                            submitBtn.innerHTML =
                                'Ajukan Booking <span class="material-symbols-outlined text-[20px]">arrow_forward</span>';
                        }
                    }).catch(error => {
                        console.error('Error:', error);
                        showToast('Terjadi kesalahan', 'error');
                        submitBtn.disabled = false;
                        submitBtn.innerHTML =
                            'Ajukan Booking <span class="material-symbols-outlined text-[20px]">arrow_forward</span>';
                    });
                    return;
                }
                if (metode === 'transfer') {
                    const bank = document.querySelector('input[name="bank_tujuan"]:checked')?.value;
                    if (!bank) {
                        showToast('Pilih bank/e-wallet tujuan', 'error');
                        return;
                    }
                    document.getElementById('bank_tujuan_final').value = bank;
                    currentFormData = new FormData(bookingForm);
                    showPaymentModal();
                    return;
                }
            });
        }

        // Payment form submit
        const paymentForm = document.getElementById('payment-form');
        if (paymentForm) {
            paymentForm.addEventListener('submit', function(e) {
                e.preventDefault();
                const jenis = document.getElementById('jenis_pembayaran').value;
                const buktiFile = document.getElementById('bukti_file').files[0];
                if (!buktiFile) {
                    showToast('Upload bukti pembayaran wajib diisi', 'error');
                    return;
                }
                const formData = new FormData();
                if (bookingForm) {
                    const bookingFormData = new FormData(bookingForm);
                    for (let [key, value] of bookingFormData.entries()) formData.append(key, value);
                }
                formData.set('jenis_pembayaran', jenis);
                formData.set('bank_tujuan_final', document.getElementById('bank_tujuan_final').value);
                formData.set('bukti_pembayaran', buktiFile);
                formData.set('metode_bayar', 'transfer');
                const submitBtnPayment = document.querySelector('#payment-form button[type="submit"]');
                submitBtnPayment.disabled = true;
                submitBtnPayment.innerHTML = 'Memproses...';
                closePaymentModal();
                fetch("{{ route('user.booking.store') }}", {
                    method: 'POST',
                    body: formData,
                    headers: {
                        'X-CSRF-TOKEN': document.querySelector('input[name="_token"]').value,
                        'Accept': 'application/json'
                    }
                }).then(async response => {
                    if (response.redirected) {
                        window.location.href = response.url;
                        return;
                    }
                    const data = await response.json();
                    if (!response.ok) throw data;
                    return data;
                }).then(data => {
                    if (data && data.success) {
                        showToast(data.message, 'success');
                        if (data.redirect) window.location.href = data.redirect;
                    }
                }).catch(error => {
                    console.error('Detail Error Server:', error);
                    if (error.errors) {
                        let errorMessages = [];
                        for (const [field, messages] of Object.entries(error.errors)) {
                            errorMessages.push(`${messages.join(', ')}`);
                        }
                        showToast(errorMessages.join(' | '), 'error');
                    } else if (error.error) {
                        showToast(error.error, 'error');
                    } else {
                        showToast('Terjadi kesalahan data, periksa console browser', 'error');
                    }
                }).finally(() => {
                    submitBtnPayment.disabled = false;
                    submitBtnPayment.innerHTML = 'Konfirmasi Booking';
                });
            });
        }

        // Initialize
        updateSummary();

        document.addEventListener('DOMContentLoaded', function() {
            setTimeout(function() {
                const skeleton = document.getElementById('skeleton-loader');
                const main = document.getElementById('main-content');
                if (skeleton) skeleton.style.display = 'none';
                if (main) main.style.display = 'block';
            }, 800);
        });

        if (document.querySelector('input[name="metode_bayar"]:checked')?.value === 'transfer') {
            document.getElementById('transfer-detail')?.classList.remove('hidden');
        }
    </script>

    </body>

    </html>
